===== chart_private.chart_fold_transition
CREATE OR REPLACE FUNCTION chart_private.chart_fold_transition(state bytea, p smallint, t text)\n RETURNS bytea\n LANGUAGE plpgsql\n IMMUTABLE\nAS $function$\nDECLARE\n  s bytea:=state;\n  mask integer;\n  count_value integer;\n  token_bytes bytea;\n  idx integer;\nBEGIN\n  IF s IS NULL OR octet_length(s)<>197 THEN s:=decode(repeat('00',197),'hex'); END IF;\n  count_value:=get_byte(s,2)*256+get_byte(s,3);\n  count_value:=least(count_value+1,65535);\n  s:=set_byte(s,2,(count_value/256)::integer);\n  s:=set_byte(s,3,(count_value%256)::integer);\n  IF p IS NULL OR p<0 OR p>11 OR t IS NULL OR t !~ '^[0-9a-f]{32}$' THEN RETURN set_byte(s,4,1); END IF;\n  mask:=get_byte(s,0)*256+get_byte(s,1);\n  IF (mask & (1<<p))<>0 THEN RETURN set_byte(s,4,1); END IF;\n  mask:=mask | (1<<p);\n  s:=set_byte(s,0,(mask/256)::integer);\n  s:=set_byte(s,1,(mask%256)::integer);\n  token_bytes:=decode(t,'hex');\n  FOR idx IN 0..15 LOOP s:=set_byte(s,5+p*16+idx,get_byte(token_bytes,idx)); END LOOP;\n  RETURN s;\nEND\n$function$\n
